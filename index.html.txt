  <script src="https://aframe.io/releases/1.5.0/aframe.min.js"></script>

  <script>
    // Logic điều khiển hệ thống sư phạm tương tác
    AFRAME.registerComponent('simulation-logic', {
      init: function () {
        this.caseOn = false;
        this.monitorOn = false;
        this.safeShutdownState = false;

        this.caseBtn = document.querySelector('#case-btn');
        this.caseLed = document.querySelector('#case-led');
        this.monitorBtn = document.querySelector('#monitor-btn');
        this.monitorScreen = document.querySelector('#monitor-screen');
        this.osShutdownBtn = document.querySelector('#os-shutdown');
        this.boardText = document.querySelector('#board-text');

        this.sndClick = document.querySelector('#snd-click');
        this.sndFan = document.querySelector('#snd-fan');

        // Bấm nút nguồn Case
        this.caseBtn.addEventListener('click', () => {
          this.sndClick.components.sound.playSound();

          if (!this.caseOn) {
            // Bật Case
            this.caseOn = true;
            this.caseLed.setAttribute('material', 'color', '#00FF00'); // Đèn xanh
            this.sndFan.components.sound.playSound();
            this.updateScreen();
            this.boardText.setAttribute('text', 'value', 'Thùng máy (Case) đã bật! Hãy bật nút nguồn Màn hình.');
          } else {
            // Bấm tắt Case đột ngột khi chưa tắt phần mềm
            if (!this.safeShutdownState) {
              this.boardText.setAttribute('text', {
                value: 'CẢNH BÁO: Tắt nguồn đột ngột dễ hỏng dữ liệu và thiết bị! Hãy tắt từ màn hình trước.',
                color: '#FF1744'
              });
            } else {
              this.caseOn = false;
              this.caseLed.setAttribute('material', 'color', '#263238');
              this.sndFan.components.sound.stopSound();
              this.updateScreen();
              this.boardText.setAttribute('text', {
                value: 'Quy trình hoàn tất! Em đã tắt máy tính rất an toàn.',
                color: '#00E676'
              });
            }
          }
        });

        // Bấm nút nguồn Màn hình
        this.monitorBtn.addEventListener('click', () => {
          this.sndClick.components.sound.playSound();
          this.monitorOn = !this.monitorOn;
          this.updateScreen();
        });

        // Thao tác Shut Down chuẩn trên màn hình Desktop ảo
        this.osShutdownBtn.addEventListener('click', () => {
          this.safeShutdownState = true;
          this.osShutdownBtn.setAttribute('visible', 'false');
          this.boardText.setAttribute('text', {
            value: 'Hệ điều hành đang lưu dữ liệu và tắt...\nBây giờ em có thể bấm nút tắt màn hình!',
            color: '#FFD54F'
          });

          setTimeout(() => {
            this.caseOn = false;
            this.caseLed.setAttribute('material', 'color', '#263238');
            this.sndFan.components.sound.stopSound();
            this.updateScreen();
          }, 2500);
        });
      },

      updateScreen: function () {
        if (this.caseOn && this.monitorOn) {
          this.monitorScreen.setAttribute('material', 'color', '#0288D1');
          this.osShutdownBtn.setAttribute('visible', 'true');
        } else if (!this.caseOn && this.monitorOn) {
          this.monitorScreen.setAttribute('material', 'color', '#37474F'); // Màn hình sáng mờ không tín hiệu
          this.osShutdownBtn.setAttribute('visible', 'false');
        } else {
          this.monitorScreen.setAttribute('material', 'color', '#000000'); // Tắt tối đen
          this.osShutdownBtn.setAttribute('visible', 'false');
        }
      }
    });
  </script>
